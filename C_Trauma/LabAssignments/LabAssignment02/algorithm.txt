#include <stdio.h>

//count digits
int countdigits(int n){
    int count=0;

    while (n>0){
        count=count+1;
        n=n/10;
    }

    return count;
}

//sum of digits
int sumofdigits(int n){

    int buffer=0;
    int a=0;
    while (n>0){
        a=n%10;
        buffer=buffer+a;
        n=n/10;
    }
    return buffer;
}

//product
int product(int n){
    int buffer=1;
    int b=0;
    while (n>0){
        b=n%10;
        buffer=buffer*b;
        n=n/10;
    }
    return buffer;
}

//largest digit
int largest(int n){
    int buf=0;
    int digit;
    while (n>0){
        digit=n%10;
        if (digit>buf){
            buf = digit;
        }
        n=n/10;
    }
    return buf;
}

//smallest digit
int smallest(int n){
    int buf=9;
    int digit;
    while (n>0){
        digit=n%10;
        if (digit<buf){
            buf = digit;
        }
        n=n/10;
    }
    return buf;
}

        n=n/10;

    return buffer;

int power(int n,int exp){
    int res=1;
    while (exp>0){
        res=res*n;
        exp=exp-1;
    }
    return res;
}

int isarm(int n)
{
    int original = n;
    int digit, sum = 0;
    int digits = count(n);

    while (n > 0)
    {
        digit = n % 10;
        sum = sum + power(digit, digits);
        n = n / 10;
    }

    return sum == original;
}

void reverse(int a[], int n)
{
    int i, temp;

    for (i = 0; i < n / 2; i++)
    {
        temp = a[i];
        a[i] = a[n - 1 - i];
        a[n - 1 - i] = temp;
    }
}

int isPrime(int n)
{
    int i;

    if (n < 2)
        return 0;

    for (i = 2; i * i <= n; i++)
    {
        if (n % i == 0)
            return 0;
    }

    return 1;
}

        }

#include <stdio.h>

void reverse(int a[], int x, int y)
{
    int temp;

    while (x < y)
    {
        temp = a[x];
        a[x] = a[y];
        a[y] = temp;

        x++;
        y--;
    }
}

int removeDup(int a[], int n)
{
    int i, j, k;

    for (i = 0; i < n; i++)
    {
        for (j = i + 1; j < n; )
        {
            if (a[i] == a[j])
            {
                for (k = j; k < n - 1; k++)
                    a[k] = a[k + 1];

int palindrome(char str[])

        end++;

    while (start < end)

        end--;
    }

#include <stdio.h>

int compare(char a[], char b[])
{
    int i = 0;

    while (a[i] != '\0' && b[i] != '\0')
    {
        if (a[i] != b[i])
            return a[i] - b[i];

        i++;
    }

    return a[i] - b[i];
}

void swap(char a[], char b[])
{
    char temp[100];
    int i;

    for (i = 0; i < 100; i++)
    {
        temp[i] = a[i];
        a[i] = b[i];
        b[i] = temp[i];

        if (temp[i] == '\0' && a[i] == '\0' && b[i] == '\0')
            break;
    }
}

void sortArray(int *p, int n)
{
    int i, j, temp;

    for (i = 0; i < n - 1; i++)
    {
        for (j = 0; j < n - 1 - i; j++)
        {
            if (*(p + j) > *(p + j + 1))
            {
                temp = *(p + j);
                *(p + j) = *(p + j + 1);
                *(p + j + 1) = temp;
            }
        }
    }
}

void analyse(int a[], int n,
             int *max, int *min,
             int *sum, float *avg)
{
    int i;

    *max = a[0];
    *min = a[0];
    *sum = 0;

    for (i = 0; i < n; i++)
    {
        *sum += a[i];

        if (a[i] > *max)
            *max = a[i];

        if (a[i] < *min)
            *min = a[i];
    }

    *avg = (float)(*sum) / n;
}

#include <stdio.h>

#define SUBJECTS 5
#define MAX_STUDENTS 100

int calculateTotal(int marks[], int n)
{
    int i, total = 0;

    for (i = 0; i < n; i++)
        total += marks[i];

    return total;
}

float calculateAverage(int marks[], int n)
{
    return (float)calculateTotal(marks, n) / n;
}

int findTopper(int totals[], int n)
{
    int i, topper = 0;

    for (i = 1; i < n; i++)
    {
        if (totals[i] > totals[topper])
            topper = i;
    }

    return topper;
}

int subjectMaximum(int marks[][SUBJECTS], int n, int sub)
{
    int i, max = marks[0][sub];

    for (i = 1; i < n; i++)
    {
        if (marks[i][sub] > max)
            max = marks[i][sub];
    }

    return max;
}

float subjectAverage(int marks[][SUBJECTS], int n, int sub)
{
    int i, sum = 0;

    for (i = 0; i < n; i++)
        sum += marks[i][sub];

    return (float)sum / n;
}

int main()
{
    int marks[MAX_STUDENTS][SUBJECTS];
    int totals[MAX_STUDENTS];
    float averages[MAX_STUDENTS];
    int n, i, j, topper;
    int allPass, passCount = 0, failCount = 0;

    printf("Enter number of students: ");
    scanf("%d", &n);

    if (n < 1 || n > MAX_STUDENTS)
        return 1;

    for (i = 0; i < n; i++)
    {
        printf("Enter marks for student %d (5 subjects): ", i + 1);

        for (j = 0; j < SUBJECTS; j++)
        {
            scanf("%d", &marks[i][j]);

            if (marks[i][j] < 0 || marks[i][j] > 100)
            {
                printf("Invalid marks\n");
                return 1;
            }
        }

        totals[i] = calculateTotal(marks[i], SUBJECTS);
        averages[i] = calculateAverage(marks[i], SUBJECTS);
    }

    printf("\nStudent Results:\n");

    for (i = 0; i < n; i++)
    {
        printf("Student %d: Total = %d, Average = %.2f\n",
               i + 1, totals[i], averages[i]);
    }

    topper = findTopper(totals, n);

    printf("\nOverall topper: Student %d\n", topper + 1);
    printf("Topper total = %d\n", totals[topper]);

    for (j = 0; j < SUBJECTS; j++)
    {
        printf("Subject %d: Maximum = %d, Average = %.2f\n",
               j + 1,
               subjectMaximum(marks, n, j),
               subjectAverage(marks, n, j));
    }

    for (i = 0; i < n; i++)
    {
        allPass = 1;

        for (j = 0; j < SUBJECTS; j++)
        {
            if (marks[i][j] < 50)
                allPass = 0;
        }

        if (allPass)
            passCount++;
        else
            failCount++;
    }

    printf("Students scoring 50 or more in all subjects = %d\n",
           passCount);
    printf("Students failing at least one subject = %d\n",
           failCount);

    return 0;
}
